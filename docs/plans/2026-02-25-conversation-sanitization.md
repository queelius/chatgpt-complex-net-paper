# Conversation Dataset Sanitization Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Produce a sanitized, research-quality public release of 1,908 ChatGPT conversations with PII redacted and an auditable redaction manifest.

**Architecture:** Three-phase pipeline: (1) Python script handles deterministic redactions (regex patterns + metadata stripping + whitelist), (2) Claude reads all conversations in parallel batches to extract third-party person names, (3) Python script applies name redactions and triages flagged files. All redactions logged to a manifest for auditability.

**Tech Stack:** Python 3.12 (stdlib only: json, re, os, shutil, hashlib, pathlib), Claude Code subagents for name extraction.

```python
#!/usr/bin/env python3
"""Sanitize ChatGPT conversation dataset for public release.

Phase 1: Deterministic regex redaction (emails, phones, paths, credentials, financial).
Phase 2: Name redaction from external manifest (generated by Claude).
Phase 3: Triage flagged files to holding bin.

Usage:
    # Phase 1: regex redaction
    python sanitize_conversations.py regex \
        --input-dir ../dev/chatgpt-4-11-2025_json_no_embeddings \
        --output-dir ../data/conversations

    # Phase 2: apply name manifest
    python sanitize_conversations.py names \
        --conversations-dir ../data/conversations \
        --manifest names-manifest.json

    # Phase 3: triage
    python sanitize_conversations.py triage \
        --conversations-dir ../data/conversations \
        --held-dir ../data/conversations-held

    # Report: summary statistics
    python sanitize_conversations.py report \
        --conversations-dir ../data/conversations
"""

import argparse
import json
import os
import re
import shutil
import hashlib
from pathlib import Path

# === WHITELIST CONFIG ===

WHITELIST_EMAILS = {
    'atowell@siue.edu',
    'queelius@gmail.com',
    'lex@metafunctor.com',
}

WHITELIST_NAMES = {
    'alexander', 'alex', 'richard', 'towell', 'queelius', 'spinoza',
}

WHITELIST_DOMAINS = {
    'metafunctor.com', 'queelius.github.io', 'github.com/queelius',
    'queelius.bsky.social', 'mastodon.social/@queelius',
    'pypi.org/user/queelius', 'queelius.r-universe.dev',
}

WHITELIST_PATTERNS = [
    re.compile(r'0000-0001-6443-9897'),  # ORCID
    re.compile(r'\bSIUE?\b'),
    re.compile(r'Southern Illinois University Edwardsville', re.I),
    re.compile(r'\bEdwardsville\b', re.I),
]

HELD_FILES = {
    'ai-research-lab-resume.json',
    'magicai-magic-ai-interview-deep-dive.json',
    'resume-builder.json',
    'resume-improvement-suggestions.json',
}

# === REGEX PATTERNS ===

EMAIL_RE = re.compile(
    r'[a-zA-Z0-9_.+-]+@[a-zA-Z0-9-]+\.[a-zA-Z]{2,}'
)

PHONE_RE = re.compile(
    r'(?<!\d)(\+?1[-.\s]?)?(\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4})(?!\d)'
)

HOME_DIR_RE = re.compile(
    r'/home/[a-zA-Z0-9_]+(?:/[^\s,;:\"\')\]}>]+)?'
)

# API keys, tokens, passwords in common formats
CREDENTIAL_RE = re.compile(
    r'(?:'
    r'(?:sk|pk|api[_-]?key|token|secret|password|bearer)[-_:]?\s*'
    r'[=:]\s*["\']?[A-Za-z0-9_\-/.+]{20,}["\']?'
    r'|'
    r'\b(?:sk-[A-Za-z0-9]{20,})'  # OpenAI-style keys
    r'|'
    r'\b(?:ghp_[A-Za-z0-9]{36,})'  # GitHub PATs
    r'|'
    r'\b(?:gho_[A-Za-z0-9]{36,})'  # GitHub OAuth
    r'|'
    r'\b(?:xox[bpas]-[A-Za-z0-9-]+)'  # Slack tokens
    r')',
    re.I
)

# Financial: account numbers, routing numbers (but not dollar amounts in
# technical contexts like "$100 per month for hosting")
FINANCIAL_RE = re.compile(
    r'(?:'
    r'\b(?:account|routing|acct)[\s#:]*\d{6,}\b'
    r'|'
    r'\b\d{4}[-\s]?\d{4}[-\s]?\d{4}[-\s]?\d{4}\b'  # credit card patterns
    r')',
    re.I
)


def is_whitelisted_email(email: str) -> bool:
    return email.lower() in {e.lower() for e in WHITELIST_EMAILS}


def redact_message(text: str, redaction_log: list, filename: str,
                   msg_idx: int, role: str) -> str:
    """Apply all regex-based redactions to a message. Returns redacted text."""
    if not text:
        return text

    def log_redaction(category, original, start, end):
        redaction_log.append({
            'file': filename,
            'message_index': msg_idx,
            'role': role,
            'category': category,
            'start': start,
            'end': end,
            'original_hash': hashlib.sha256(original.encode()).hexdigest()[:16],
            'length': len(original),
        })

    # Emails (skip whitelisted)
    for m in reversed(list(EMAIL_RE.finditer(text))):
        email = m.group()
        if not is_whitelisted_email(email):
            log_redaction('EMAIL', email, m.start(), m.end())
            text = text[:m.start()] + '[EMAIL]' + text[m.end():]

    # Phone numbers
    for m in reversed(list(PHONE_RE.finditer(text))):
        phone = m.group()
        log_redaction('PHONE', phone, m.start(), m.end())
        text = text[:m.start()] + '[PHONE]' + text[m.end():]

    # Home directory paths
    for m in reversed(list(HOME_DIR_RE.finditer(text))):
        path = m.group()
        # Skip whitelisted usernames
        if any(f'/home/{name}' in path.lower() for name in ['spinoza', 'alex']):
            # Still redact the path but not as PII — these are system paths
            log_redaction('HOME_DIR', path, m.start(), m.end())
            text = text[:m.start()] + '[HOME_DIR]' + text[m.end():]

    # Credentials
    for m in reversed(list(CREDENTIAL_RE.finditer(text))):
        cred = m.group()
        log_redaction('CREDENTIAL', cred, m.start(), m.end())
        text = text[:m.start()] + '[CREDENTIAL]' + text[m.end():]

    # Financial
    for m in reversed(list(FINANCIAL_RE.finditer(text))):
        fin = m.group()
        log_redaction('FINANCIAL', fin, m.start(), m.end())
        text = text[:m.start()] + '[FINANCIAL]' + text[m.end():]

    return text


def cmd_regex(args):
    """Phase 1: Copy files and apply regex-based redactions."""
    input_dir = Path(args.input_dir)
    output_dir = Path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    files = sorted(f for f in os.listdir(input_dir) if f.endswith('.json'))
    redaction_log = []
    stats = {'files': 0, 'messages': 0, 'redactions': {}}

    for filename in files:
        with open(input_dir / filename) as fh:
            conv = json.load(fh)

        # Strip openai_url
        if 'openai_url' in conv:
            del conv['openai_url']

        # Redact messages
        for i, msg in enumerate(conv.get('messages', [])):
            content = msg.get('content', '') or ''
            msg['content'] = redact_message(
                content, redaction_log, filename, i, msg.get('role', 'unknown')
            )
            stats['messages'] += 1

        # Write sanitized file
        with open(output_dir / filename, 'w') as fh:
            json.dump(conv, fh, indent=2, ensure_ascii=False)

        stats['files'] += 1

    # Count redactions by type
    for entry in redaction_log:
        cat = entry['category']
        stats['redactions'][cat] = stats['redactions'].get(cat, 0) + 1

    # Save redaction log
    log_path = output_dir / 'redaction-log-regex.json'
    with open(log_path, 'w') as fh:
        json.dump({
            'phase': 'regex',
            'total_redactions': len(redaction_log),
            'stats': stats,
            'entries': redaction_log,
        }, fh, indent=2)

    print(f"Phase 1 complete: {stats['files']} files, "
          f"{stats['messages']} messages, "
          f"{len(redaction_log)} redactions")
    for cat, count in sorted(stats['redactions'].items()):
        print(f"  {cat}: {count}")


def cmd_names(args):
    """Phase 2: Apply name redactions from Claude-generated manifest."""
    conv_dir = Path(args.conversations_dir)
    manifest_path = Path(args.manifest)

    with open(manifest_path) as fh:
        manifest = json.load(fh)

    redaction_log = []
    files_modified = 0

    for entry in manifest.get('names', []):
        filename = entry['file']
        names_to_redact = entry['names']
        if not names_to_redact:
            continue

        filepath = conv_dir / filename
        if not filepath.exists():
            continue

        with open(filepath) as fh:
            conv = json.load(fh)

        modified = False
        for i, msg in enumerate(conv.get('messages', [])):
            content = msg.get('content', '') or ''
            for name in sorted(names_to_redact, key=len, reverse=True):
                # Case-insensitive replacement
                pattern = re.compile(re.escape(name), re.I)
                matches = list(pattern.finditer(content))
                if matches:
                    modified = True
                    for m in reversed(matches):
                        redaction_log.append({
                            'file': filename,
                            'message_index': i,
                            'role': msg.get('role', 'unknown'),
                            'category': 'PERSON',
                            'start': m.start(),
                            'end': m.end(),
                            'original_hash': hashlib.sha256(
                                m.group().encode()).hexdigest()[:16],
                            'length': len(m.group()),
                        })
                        content = content[:m.start()] + '[PERSON]' + content[m.end():]
                msg['content'] = content

        if modified:
            with open(filepath, 'w') as fh:
                json.dump(conv, fh, indent=2, ensure_ascii=False)
            files_modified += 1

    # Save name redaction log
    log_path = conv_dir / 'redaction-log-names.json'
    with open(log_path, 'w') as fh:
        json.dump({
            'phase': 'names',
            'total_redactions': len(redaction_log),
            'files_modified': files_modified,
            'entries': redaction_log,
        }, fh, indent=2)

    print(f"Phase 2 complete: {files_modified} files modified, "
          f"{len(redaction_log)} name redactions")


def cmd_triage(args):
    """Phase 3: Move held files to holding bin."""
    conv_dir = Path(args.conversations_dir)
    held_dir = Path(args.held_dir)
    held_dir.mkdir(parents=True, exist_ok=True)

    moved = []

    # Move pre-identified held files
    for filename in HELD_FILES:
        src = conv_dir / filename
        if src.exists():
            shutil.move(str(src), str(held_dir / filename))
            moved.append(filename)

    # Move any files flagged in the name manifest
    manifest_path = conv_dir / 'redaction-log-names.json'
    if manifest_path.exists():
        with open(manifest_path) as fh:
            log = json.load(fh)
        # Files with many name redactions might need manual review
        name_counts = {}
        for entry in log.get('entries', []):
            f = entry['file']
            name_counts[f] = name_counts.get(f, 0) + 1
        # Threshold: files with 10+ name redactions go to review
        for f, count in name_counts.items():
            if count >= 10 and f not in HELD_FILES:
                src = conv_dir / f
                if src.exists():
                    shutil.move(str(src), str(held_dir / f))
                    moved.append(f)

    print(f"Phase 3 complete: {len(moved)} files moved to holding bin")
    for f in moved:
        print(f"  {f}")


def cmd_report(args):
    """Print summary statistics for the sanitized dataset."""
    conv_dir = Path(args.conversations_dir)

    files = [f for f in os.listdir(conv_dir)
             if f.endswith('.json') and not f.startswith('redaction-log')]
    total_size = sum(os.path.getsize(conv_dir / f) for f in files)
    total_messages = 0
    for f in files:
        with open(conv_dir / f) as fh:
            conv = json.load(fh)
        total_messages += len(conv.get('messages', []))

    print(f"Sanitized dataset:")
    print(f"  Files: {len(files)}")
    print(f"  Messages: {total_messages}")
    print(f"  Size: {total_size / 1024 / 1024:.1f} MB")

    # Merge redaction logs
    total_redactions = 0
    by_category = {}
    for log_name in ['redaction-log-regex.json', 'redaction-log-names.json']:
        log_path = conv_dir / log_name
        if log_path.exists():
            with open(log_path) as fh:
                log = json.load(fh)
            total_redactions += log.get('total_redactions', 0)
            if 'stats' in log:
                for cat, count in log['stats'].get('redactions', {}).items():
                    by_category[cat] = by_category.get(cat, 0) + count
            # Count PERSON from names log
            for entry in log.get('entries', []):
                if entry['category'] == 'PERSON':
                    by_category['PERSON'] = by_category.get('PERSON', 0) + 1

    print(f"  Total redactions: {total_redactions}")
    for cat, count in sorted(by_category.items()):
        print(f"    {cat}: {count}")


def main():
    parser = argparse.ArgumentParser(description='Sanitize conversation dataset')
    subparsers = parser.add_subparsers(dest='command', required=True)

    # Phase 1: regex
    p_regex = subparsers.add_parser('regex', help='Apply regex-based redactions')

    p_regex.add_argument('--output-dir', required=True)

    # Phase 2: names

    p_names.add_argument('--conversations-dir', required=True)
    p_names.add_argument('--manifest', required=True)

    # Phase 3: triage
    p_triage = subparsers.add_parser('triage', help='Move flagged files to holding bin')
    p_triage.add_argument('--conversations-dir', required=True)
    p_triage.add_argument('--held-dir', required=True)

    # Report
    p_report = subparsers.add_parser('report', help='Print summary statistics')
    p_report.add_argument('--conversations-dir', required=True)

    args = parser.parse_args()

    commands = {
        'regex': cmd_regex,
        'names': cmd_names,
        'triage': cmd_triage,
        'report': cmd_report,
    }
    commands[args.command](args)


if __name__ == '__main__':
    main()
```

1. Reads all messages in the batch
2. Identifies every person name that is NOT on the whitelist
3. Returns a list of `{file, names: [list of name strings]}` entries
4. Flags files with unusually high PII density
